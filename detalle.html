// Diagrama de pines (SIM800L ↔ Arduino)

SIM800L        Arduino
────────────────────────
VCC  ──────→  LM2596 (4V OUT)
GND  ──────→  GND
TXD  ──────→  PIN 10 (SoftSerial RX)
RXD  ──────→  PIN 11 (SoftSerial TX)
RST  ──────→  PIN 12 (opcional)

MIC+ ──────→  Micrófono (+)
MIC- ──────→  GND
SPK+ ──────→  Altavoz (+)
SPK- ──────→  Altavoz (−)

Botón: PIN 2 → 10kΩ → GND

⚠ Advertencias

  • El SIM800L consume hasta 2A en picos durante llamadas.
  • Nunca alimentar desde el pin 5V del Arduino (máx. 500mA).
  • El voltaje ideal del módulo es 3.7–4.2V.
  • El ruido "clic-clic" viene de la interferencia RF del GSM con el audio.

// Sketch Arduino

Código Fuente Completo

Copia este sketch en el IDE de Arduino. Ajusta NUMERO_DESTINO y los pines según tu conexión.

#include <SoftwareSerial.h>

// ─── Configuración de pines ──────────────────────
#define PIN_TX      11      // Arduino TX → SIM800L RXD
#define PIN_RX      10      // Arduino RX ← SIM800L TXD
#define PIN_BOTON    2      // Botón de llamada
#define PIN_LED     13      // LED de estado (built-in)

// ─── Número de destino ───────────────────────────
const char* NUMERO_DESTINO = "+5411XXXXXXXX";  // ← CAMBIAR

// ─── Software Serial ─────────────────────────────
SoftwareSerial gsm(PIN_RX, PIN_TX);

// ─── Variables de estado ─────────────────────────
bool enLlamada     = false;
bool botonAnterior = HIGH;

// ─────────────────────────────────────────────────
void setup() {
  pinMode(PIN_BOTON, INPUT_PULLUP);
  pinMode(PIN_LED,   OUTPUT);

  Serial.begin(9600);
  gsm.begin(9600);

  Serial.println("Iniciando modulo GSM...");
  delay(3000);               // Esperar que el modulo arranque

  iniciarGSM();
}

// ─────────────────────────────────────────────────
void loop() {
  bool botonActual = digitalRead(PIN_BOTON);

  // Detectar flanco descendente (botón presionado)
  if (botonAnterior == HIGH && botonActual == LOW) {
    delay(50);  // Debounce

    if (!enLlamada) {
      hacerLlamada();
    } else {
      colgarLlamada();
    }
  }

  botonAnterior = botonActual;

  // Leer respuestas del modulo (para debug)
  while (gsm.available()) {
    Serial.write(gsm.read());
  }
}

// ─────────────────────────────────────────────────
void iniciarGSM() {
  enviarComando("AT",        "OK", 1000);   // Verificar comunicacion
  enviarComando("ATE0",      "OK", 1000);   // Desactivar eco
  enviarComando("AT+CMGF=1", "OK", 1000);   // Modo texto SMS
  Serial.println("SIM800L listo.");
}

// ─────────────────────────────────────────────────
void hacerLlamada() {
  Serial.print("Llamando a: ");
  Serial.println(NUMERO_DESTINO);

  String cmd = "ATD";
  cmd += NUMERO_DESTINO;
  cmd += ";";           // Punto y coma = llamada de VOZ
  gsm.println(cmd);

  enLlamada = true;
  digitalWrite(PIN_LED, HIGH);
}

// ─────────────────────────────────────────────────
void colgarLlamada() {
  Serial.println("Colgando...");
  gsm.println("ATH");

  enLlamada = false;
  digitalWrite(PIN_LED, LOW);
}

// ─────────────────────────────────────────────────
void enviarComando(const char* cmd, const char* respuesta, int timeout) {
  gsm.println(cmd);
  long inicio = millis();
  String buffer = "";

  while (millis() - inicio < timeout) {
    while (gsm.available()) {
      buffer += (char)gsm.read();
    }
    if (buffer.indexOf(respuesta) != -1) break;
  }

  Serial.print("[GSM] ");
  Serial.println(buffer);
}

// Flujo de estados

[IDLE]
  ↓ botón presionado
[LLAMANDO] → ATD+nro;
  ↓ conexión establecida
[EN LLAMADA]
  ↓ botón presionado
[COLGANDO] → ATH
  ↓
[IDLE]

Ver Proyecto en Tinkercad

El circuito completo está disponible en Tinkercad Circuits. Podés abrir la simulación en tu navegador sin instalar nada. Tinkercad no soporta el módulo SIM800L real, por lo que la simulación muestra la lógica del Arduino y el botón en un entorno equivalente.

// Qué simular

  • Lógica del botón (debounce incluido)
  • Comunicación SoftwareSerial
  • LED de estado de llamada
  • Monitor Serial con respuestas AT

// Limitaciones de Tinkercad

  • SIM800L no disponible en la librería
  • No simula llamadas de voz reales

  • La lógica de control sí es 100% válida
